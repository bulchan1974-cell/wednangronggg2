SQL Schema สำหรับสร้างตาราง

นำไปวางและกด Run ในเมนู SQL Editor ของ Supabase

create table if not exists public.student_registrations (
  id uuid primary key default gen_random_uuid(),
  student_code text not null unique,
  education_level text not null,
  department text not null,
  prefix text not null,
  full_name text not null,
  national_id text not null unique,
  birth_date date not null,
  phone text not null,
  email text not null,
  address text not null,
  guardian_name text not null,
  guardian_phone text not null,
  created_at timestamptz not null default now()
);

alter table public.student_registrations enable row level security;

create policy "Allow anon select student_registrations"
on public.student_registrations for select to anon using (true);

create policy "Allow anon insert student_registrations"
on public.student_registrations for insert to anon with check (true);

create policy "Allow anon delete student_registrations"
on public.student_registrations for delete to anon using (true);
